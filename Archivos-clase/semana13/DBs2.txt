-- -----------------------------------------------------
-- Script para crear la Base de Datos: gestion_tesis_db
-- Basado en los 9 modelos Java (MVC)
-- -----------------------------------------------------

-- 1. CREACIÓN DE LA BASE DE DATOS
DROP SCHEMA IF EXISTS `gestion_tesis_db`;
CREATE SCHEMA IF NOT EXISTS `gestion_tesis_db` DEFAULT CHARACTER SET utf8mb4 ;
USE `gestion_tesis_db` ;

-- -----------------------------------------------------
-- Tabla 1: `usuarios` (Modelo: Usuario.java)
-- Tabla base que almacena los datos comunes de login.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `usuarios` (
  `idUsuario` INT NOT NULL AUTO_INCREMENT,
  `nombreUsuario` VARCHAR(50) NOT NULL,
  `contrasena` VARCHAR(255) NOT NULL, -- Debe ser larga para el hash (SHA-256)
  `nombre` VARCHAR(100) NOT NULL,
  `apellido` VARCHAR(100) NOT NULL,
  `email` VARCHAR(100) NOT NULL,
  `rol` ENUM('estudiante', 'docente', 'admin') NOT NULL,
  `fotoPerfilUrl` VARCHAR(255) NULL DEFAULT 'default.png',
  `activo` TINYINT(1) NOT NULL DEFAULT 1, -- 1=Activo, 0=Inactivo
  PRIMARY KEY (`idUsuario`),
  UNIQUE INDEX `email_UNIQUE` (`email` ASC),
  UNIQUE INDEX `nombreUsuario_UNIQUE` (`nombreUsuario` ASC)
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 2: `estudiantes` (Modelo: Estudiante.java)
-- Extiende de 'usuarios'. Almacena datos solo de estudiantes.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `estudiantes` (
  `idEstudiante` INT NOT NULL, -- Es el mismo ID que idUsuario (Llave Primaria y Foránea)
  `codigoEstudiante` VARCHAR(20) NOT NULL,
  `carrera` VARCHAR(100) NULL,
  PRIMARY KEY (`idEstudiante`),
  UNIQUE INDEX `codigoEstudiante_UNIQUE` (`codigoEstudiante` ASC),
  CONSTRAINT `fk_estudiantes_usuarios`
    FOREIGN KEY (`idEstudiante`)
    REFERENCES `usuarios` (`idUsuario`)
    ON DELETE CASCADE -- Si se borra el usuario, se borra el estudiante
    ON UPDATE CASCADE
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 3: `docentes` (Modelo: Docente.java)
-- Extiende de 'usuarios'. Almacena datos solo de docentes.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `docentes` (
  `idDocente` INT NOT NULL, -- Es el mismo ID que idUsuario
  `departamento` VARCHAR(100) NULL,
  `especialidad` VARCHAR(100) NULL,
  PRIMARY KEY (`idDocente`),
  CONSTRAINT `fk_docentes_usuarios`
    FOREIGN KEY (`idDocente`)
    REFERENCES `usuarios` (`idUsuario`)
    ON DELETE CASCADE
    ON UPDATE CASCADE
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 4: `tesis` (Modelo: Tesis.java)
-- La tabla principal que conecta todo.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `tesis` (
  `idTesis` INT NOT NULL AUTO_INCREMENT,
  `titulo` VARCHAR(255) NOT NULL,
  `descripcion` TEXT NULL,
  `fechaSubida` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `estado` ENUM('Pendiente', 'En Revisión', 'Requiere Correcciones', 'Aprobado', 'Rechazado') NOT NULL DEFAULT 'Pendiente',
  `urlArchivo` VARCHAR(255) NULL,
  `fechaProximaRevision` DATE NULL, -- Para el calendario y panel del docente
  `idEstudianteAutor` INT NOT NULL,
  `idDocenteRevisor` INT NULL, -- Es NULLABLE, porque el admin lo asigna después
  PRIMARY KEY (`idTesis`),
  INDEX `fk_tesis_estudiantes_idx` (`idEstudianteAutor` ASC),
  INDEX `fk_tesis_docentes_idx` (`idDocenteRevisor` ASC),
  CONSTRAINT `fk_tesis_estudiantes`
    FOREIGN KEY (`idEstudianteAutor`)
    REFERENCES `estudiantes` (`idEstudiante`)
    ON DELETE NO ACTION -- No dejar borrar un estudiante si tiene tesis
    ON UPDATE CASCADE,
  CONSTRAINT `fk_tesis_docentes`
    FOREIGN KEY (`idDocenteRevisor`)
    REFERENCES `docentes` (`idDocente`)
    ON DELETE SET NULL -- Si se borra el docente, la tesis queda "Sin Asignar"
    ON UPDATE CASCADE
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 5: `comentarios` (Modelo: Comentario.java)
-- Almacena los comentarios de revisión del docente.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `comentarios` (
  `idComentario` INT NOT NULL AUTO_INCREMENT,
  `texto` TEXT NOT NULL,
  `fechaHora` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `idTesis` INT NOT NULL,
  `idDocenteAutor` INT NOT NULL,
  PRIMARY KEY (`idComentario`),
  INDEX `fk_comentarios_tesis_idx` (`idTesis` ASC),
  INDEX `fk_comentarios_docentes_idx` (`idDocenteAutor` ASC),
  CONSTRAINT `fk_comentarios_tesis`
    FOREIGN KEY (`idTesis`)
    REFERENCES `tesis` (`idTesis`)
    ON DELETE CASCADE -- Si se borra la tesis, se borran sus comentarios
    ON UPDATE CASCADE,
  CONSTRAINT `fk_comentarios_docentes`
    FOREIGN KEY (`idDocenteAutor`)
    REFERENCES `docentes` (`idDocente`)
    ON DELETE NO ACTION -- No dejar borrar al docente si tiene comentarios
    ON UPDATE CASCADE
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 6: `mensajes` (Modelo: Mensaje.java)
-- Para la mensajería interna entre usuarios.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mensajes` (
  `idMensaje` INT NOT NULL AUTO_INCREMENT,
  `asunto` VARCHAR(255) NULL,
  `cuerpo` TEXT NULL,
  `fechaHoraEnvio` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `leido` TINYINT(1) NOT NULL DEFAULT 0,
  `idRemitente` INT NOT NULL,
  `idDestinatario` INT NOT NULL,
  PRIMARY KEY (`idMensaje`),
  INDEX `fk_mensajes_remitente_idx` (`idRemitente` ASC),
  INDEX `fk_mensajes_destinatario_idx` (`idDestinatario` ASC),
  CONSTRAINT `fk_mensajes_remitente`
    FOREIGN KEY (`idRemitente`)
    REFERENCES `usuarios` (`idUsuario`)
    ON DELETE NO ACTION
    ON UPDATE CASCADE,
  CONSTRAINT `fk_mensajes_destinatario`
    FOREIGN KEY (`idDestinatario`)
    REFERENCES `usuarios` (`idUsuario`)
    ON DELETE NO ACTION
    ON UPDATE CASCADE
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 7: `evento_calendario` (Modelo: EventoCalendario.java)
-- Para eventos generales del calendario (ej. "Sustentaciones").
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `evento_calendario` (
  `idEvento` INT NOT NULL AUTO_INCREMENT,
  `titulo` VARCHAR(100) NOT NULL,
  `descripcion` TEXT NULL,
  `fechaEvento` DATE NOT NULL,
  `tipo` VARCHAR(50) NULL DEFAULT 'General',
  PRIMARY KEY (`idEvento`)
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 8: `historial_estado` (Modelo: HistorialEstado.java)
-- Guarda el "timeline" de una tesis.
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `historial_estado` (
  `idHistorial` INT NOT NULL AUTO_INCREMENT,
  `fechaCambio` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `estadoNuevo` VARCHAR(50) NOT NULL,
  `notas` VARCHAR(255) NULL,
  `idTesis` INT NOT NULL,
  `idUsuarioModificador` INT NOT NULL,
  PRIMARY KEY (`idHistorial`),
  INDEX `fk_historial_tesis_idx` (`idTesis` ASC),
  INDEX `fk_historial_usuarios_idx` (`idUsuarioModificador` ASC),
  CONSTRAINT `fk_historial_tesis`
    FOREIGN KEY (`idTesis`)
    REFERENCES `tesis` (`idTesis`)
    ON DELETE CASCADE -- Si se borra la tesis, se borra su historial
    ON UPDATE CASCADE,
  CONSTRAINT `fk_historial_usuarios`
    FOREIGN KEY (`idUsuarioModificador`)
    REFERENCES `usuarios` (`idUsuario`)
    ON DELETE NO ACTION
    ON UPDATE CASCADE
) ENGINE = InnoDB;

-- -----------------------------------------------------
-- Tabla 9: `configuracion_sistema` (Modelo: ConfiguracionSistema.java)
-- Guarda variables globales del sistema (logo, periodo, etc.)
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `configuracion_sistema` (
  `idConfig` INT NOT NULL DEFAULT 1,
  `nombreInstitucion` VARCHAR(255) NULL,
  `logoUrl` VARCHAR(255) NULL DEFAULT 'logo_default.png',
  `periodoAcademico` VARCHAR(50) NULL,
  PRIMARY KEY (`idConfig`)
) ENGINE = InnoDB;
